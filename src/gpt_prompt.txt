def _kdf_stream(_key: bytes, _nonce: bytes, _n: int) -> bytes:
    """Derive a pseudorandom stream of length n using BLAKE2b."""
    out = bytearray()
    ctr = 0
    while len(out) < _n:
        h = _hashlib.blake2b(key=_key, person=_nonce[:16], digest_size=32)
        h.update(_nonce)
        h.update(ctr.to_bytes(4, "big"))
        out.extend(h.digest())
        ctr += 1
    return bytes(out[:_n])

def encrypt_secret(app_secret: str, plaintext: str) -> str:
    """utf-8 -> 'nonce_hex:cipher_hex' (reversible, stdlib-only)."""
    nonce = _secrets.token_bytes(24)

    pt = plaintext.encode("utf-8")
    ks = _kdf_stream(key, nonce, len(pt))
    ct = bytes(a ^ b for a, b in zip(pt, ks))

def decrypt_secret(app_secret: str, blob: str) -> str:
    """'nonce_hex:cipher_hex' -> utf-8."""
